# =========================================================
# 1. تهيئة مشروع جديد وتجهيزه للرفع لأول مرة
# =========================================================

# إنشاء مستودع Git فارغ محلياً داخل مجلد المشروع
git init

# تجهيز وتتبع جميع الملفات والتعديلات في المجلد (مع استثناء ما في .gitignore)
git add .

# حفظ التعديلات المجهزة في السجل المحلي وإعطائها وصفاً (Commit)
git commit -m "Initial commit"

# تغيير اسم الفرع الافتراضي المحلي إلى main لتطابق معايير GitHub
git branch -M main

# ربط المستودع المحلي بالمستودع البعيد على GitHub باستخدام الرابط
git remote add origin https://github.com/AbdulqaderGhallab/ghallab-shop.git

# رفع الفرع الرئيسي إلى GitHub وتحديد origin/main كمسار افتراضي للرفع مستقبلاً
git push -u origin main


# =========================================================
# 2. حل أخطاء عدم التوافق والتعارض عند الرفع الأول (الأوامر التي عرقلتنا)
# =========================================================

# جلب وتجميع التغييرات من GitHub حتى لو كان للمستودعين تاريخان غير مرتبطين (تستخدم إذا كان المستودع يحتوي على ملفات سابقة كـ README)
git pull origin main --allow-unrelated-histories

# حل تعارض التعديلات في ملف محدد عبر اختيار النسخة المحلية الخاصة بك
git checkout --ours README.md

# تأكيد تجهيز الملف بعد حل التعارض فيه
git add README.md

# حفظ عملية حل التعارض ودمج التاريخين كـ Commit جديد
git commit -m "Fix conflict and merge remote"

# إعادة رفع المشروع بعد حل كافة التعارضات بنجاح
git push -u origin main


# =========================================================
# 3. ربط مشروع محلي بمستودع موجود بالفعل على GitHub
# =========================================================

# إضافة رابط السيرفر البعيد للمشروع
git remote add origin https://github.com/AbdulqaderGhallab/REPOSITORY_NAME.git

# عرض قوائم وروابط المستودعات البعيدة المربوطة للتأكد من صحتها
git remote -v

# رفع التعديلات إلى الفرع الرئيسي للمستودع المربوط
git push -u origin main


# =========================================================
# 4. إنشاء وإدارة الفروع (Branches)
# =========================================================

# إنشاء فرع جديد والتحول للعمل عليه مباشرة
git checkout -b feature-branch-name

# جلب بيانات وتحديثات كافة الفروع الموجودة على السيرفر البعيد
git fetch origin

# ربط الفرع المحلي الحالي بفرع محدد على السيرفر البعيد لتسهيل الرفع والسحب
git branch --set-upstream-to=origin/feature-branch-name feature-branch-name

# التنقل والتبديل إلى فرع موجود سابقاً
git checkout branch-name

# الأمر الحديث البديل للتبديل بين الفروع بوضوح
git switch branch-name


# =========================================================
# 5. التحديث والرفع اليومي للعمل
# =========================================================

# فحص حالة الملفات ومعرفة الملفات المعدلة أو الجديدة أو غير المتتبعة
git status

# تجهيز كافة التعديلات اليومية للـ Commit
git add .

# توثيق التعديلات اليومية برسالة واضحة تشرح ما تم إنجازه
git commit -m "Describe your daily updates"

# إرسال ورفع الـ Commits الجديدة إلى السيرفر البعيد
git push


# =========================================================
# 6. دمج الفروع محلياً (Merging)
# =========================================================

# التبديل للفرع الرئيسي تمهيداً لدمج العمل فيه
git checkout main

# سحب أحدث نسخة من الفرع الرئيسي من السيرفر لضمان عدم وجود تعارض
git pull origin main

# دمج التعديلات المنجزة من الفرع الفرعي إلى الفرع الحالي (main)
git merge feature-branch-name

# رفع الفرع الرئيسي إلى السيرفر بعد اكتمال الدمج
git push origin main


# =========================================================
# 7. أوامر الطوارئ والتراجع المتقدمة
# =========================================================

# عرض سجل التغييرات والـ Commits السابقة بشكل مختصر في خط واحد لكل تعديل
git log --oneline

# التراجع عن التعديلات غير المحفوظة في ملف محدد وإعادته لآخر حالة كان عليها
git checkout -- filename.ext

# تخزين التعديلات الحالية غير المكتملة في ذاكرة مؤقتة وتنظيف مساحة العمل
git stash

# استرجاع التعديلات التي تم تخزينها مؤقتاً للعودة للعمل عليها
git stash pop

# حذف فرع محلي بعد الانتهاء من دمجه وعدم الحاجة إليه
git branch -d feature-branch-name
